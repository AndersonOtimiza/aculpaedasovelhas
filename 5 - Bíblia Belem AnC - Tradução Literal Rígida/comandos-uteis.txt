=== COMANDOS ÚTEIS - BIBLIA CODICES AI ===

=== POSTGRESQL ===
# Criar banco de dados
createdb biblia_codices_ai

# Conectar ao banco
psql -d biblia_codices_ai

# Executar schema
psql -d biblia_codices_ai -f database/schema.sql

# Backup do banco
pg_dump biblia_codices_ai > backup.sql

# Restaurar backup
psql -d biblia_codices_ai < backup.sql

# Listar bancos
psql -l

# Deletar banco (cuidado!)
dropdb biblia_codices_ai

# Queries úteis dentro do psql
\dt                          # Listar tabelas
\d books                     # Descrever tabela books
\d+ books                    # Descrição detalhada
SELECT * FROM books;         # Ver todos os livros
SELECT COUNT(*) FROM verses; # Contar versículos
\q                           # Sair do psql

=== NPM / NODE ===
# Instalar dependências
cd backend
npm install

# Modo desenvolvimento (auto-reload)
npm run dev

# Build para produção
npm run build

# Executar versão compilada
npm start

# Adicionar nova dependência
npm install nome-do-pacote

# Adicionar dependência de desenvolvimento
npm install --save-dev nome-do-pacote

# Verificar versão do Node
node --version

# Verificar versão do npm
npm --version

=== TESTES DE API (usando curl) ===
# Health check
curl http://localhost:3000/health

# Listar livros
curl http://localhost:3000/api/books

# Buscar livro específico
curl http://localhost:3000/api/books/GEN

# Listar manuscritos
curl http://localhost:3000/api/manuscripts

# Buscar por lemma (com encoding UTF-8)
curl "http://localhost:3000/api/tokens/search/lemma?lemma=palavra"

Invoke-WebRequest -Uri "http://localhost:3000/health" | Select-Object -Expand Content

=== DOCKER (Futuro) ===

# Docker Compose (com PostgreSQL)

=== MANUTENÇÃO DO BANCO ===
# Vacuum (limpeza e otimização)
psql -d biblia_codices_ai -c "VACUUM ANALYZE;"

# Reindexar tabelas
psql -d biblia_codices_ai -c "REINDEX DATABASE biblia_codices_ai;"

# Ver tamanho do banco
psql -d biblia_codices_ai -c "SELECT pg_size_pretty(pg_database_size('biblia_codices_ai'));"

# Ver tamanho das tabelas
psql -d biblia_codices_ai -c "SELECT relname, pg_size_pretty(pg_total_relation_size(relid)) FROM pg_catalog.pg_statio_user_tables ORDER BY pg_total_relation_size(relid) DESC;"

=== TROUBLESHOOTING ===
# Porta 3000 já em uso
# Windows: netstat -ano | findstr :3000
# Matar processo: taskkill /PID <PID> /F

# PostgreSQL não conecta
# Verificar se está rodando: Get-Service postgresql*
# Iniciar serviço: Start-Service postgresql-x64-<versão>

# Erro de módulo não encontrado
# Reinstalar: rm -rf node_modules && npm install

# Erro de TypeScript
# Limpar build: rm -rf dist && npm run build

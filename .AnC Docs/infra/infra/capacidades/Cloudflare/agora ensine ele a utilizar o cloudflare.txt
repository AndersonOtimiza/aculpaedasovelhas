"""Utilitario de automacao para Cloudflare.

Este modulo substitui o guia original por uma CLI funcional que executa
operacoes da Cloudflare: busca de zona, limpeza de cache e gerenciamento de
registros DNS. As credenciais sao obtidas do 1Password quando possivel, com
fallback para flags e variaveis de ambiente.

Exemplos de uso (coloque o nome do arquivo entre aspas ao chamar diretamente):

   python "agora ensine ele a utilizar o cloudflare.py" zone --zone otimiza.pro
   python "agora ensine ele a utilizar o cloudflare.py" purge --zone otimiza.pro
   python "agora ensine ele a utilizar o cloudflare.py" dns list --zone otimiza.pro
   python "agora ensine ele a utilizar o cloudflare.py" dns create --zone otimiza.pro --type A --name portal \
      --content 200.160.1.1 --proxied

Defina estas variaveis de ambiente ou utilize o item 1Password ``Cloudflare_API``
no cofre ``OtimizaPro``:

   CLOUDFLARE_API_TOKEN
   CLOUDFLARE_API_KEY
   CLOUDFLARE_EMAIL
   CLOUDFLARE_ZONE_ID

"""
from __future__ import annotations

import argparse
import json
import logging
import os
import shutil
import subprocess
import sys
from dataclasses import dataclass
from typing import Any, Dict, Iterable, Optional, Sequence
from urllib import error as urllib_error
from urllib import parse as urllib_parse
from urllib import request as urllib_request

API_BASE_URL = "https://api.cloudflare.com/client/v4"
DEFAULT_OP_ITEM = "Cloudflare_API"
DEFAULT_OP_VAULT = "OtimizaPro"
LOG_PATH = os.path.join("logs", "cloudflare.log")


class CloudflareError(RuntimeError):
   """Excecao disparada quando a API Cloudflare retorna erro."""

   def __init__(self, status_code: int, message: str, *, payload: Optional[Dict[str, Any]] = None) -> None:
      super().__init__(f"Erro da API Cloudflare {status_code}: {message}")
      self.status_code = status_code
      self.payload = payload or {}


@dataclass
class CloudflareCredentials:
   """Agrupa dados de autenticacao e configuracoes opcionais."""

   api_token: Optional[str] = None
   api_key: Optional[str] = None
   email: Optional[str] = None
   zone_id: Optional[str] = None
   account_id: Optional[str] = None

   def headers(self) -> Dict[str, str]:
      if self.api_token:
         return {
            "Authorization": f"Bearer {self.api_token}",
            "Content-Type": "application/json",
         }
      if self.api_key and self.email:
         return {
            "X-Auth-Key": self.api_key,
            "X-Auth-Email": self.email,
            "Content-Type": "application/json",
         }
      raise ValueError("Autenticacao ausente; informe um token de API ou chave/e-mail.")


class CloudflareClient:
   """Wrapper simples sobre a API REST da Cloudflare."""

   def __init__(self, credentials: CloudflareCredentials, *, timeout: int = 30) -> None:
      self._credentials = credentials
      self._timeout = timeout

   def _request(self, method: str, path: str, *, params: Optional[Dict[str, Any]] = None,
             body: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
      url = f"{API_BASE_URL}{path}"
      if params:
         query = urllib_parse.urlencode(params, doseq=True)
         url = f"{url}?{query}"
      headers = dict(self._credentials.headers())
      data_bytes = None
      if body is not None:
         data_bytes = json.dumps(body).encode("utf-8")
         headers.setdefault("Content-Type", "application/json")
      request_obj = urllib_request.Request(url, data=data_bytes, headers=headers, method=method)
      try:
         with urllib_request.urlopen(request_obj, timeout=self._timeout) as response:
            status = response.status
            content_type = response.headers.get("Content-Type", "")
            raw = response.read()
      except urllib_error.HTTPError as exc:
         status = exc.code
         content_type = exc.headers.get("Content-Type", "") if exc.headers else ""
         raw = exc.read()
         payload = _parse_payload(raw, content_type)
         logging.error(
            "Falha na requisicao para Cloudflare",
            extra={"status": status, "payload": payload, "method": method, "url": url},
         )
         raise CloudflareError(status, _first_error(payload), payload=payload) from exc
      except urllib_error.URLError as exc:
         logging.error("Nao foi possivel contactar a Cloudflare", exc_info=exc)
         raise ConnectionError(str(exc)) from exc
      payload = _parse_payload(raw, content_type)
      if status >= 400 or not payload.get("success", True):
         logging.error(
            "Falha na requisicao para Cloudflare",
            extra={"status": status, "payload": payload, "method": method, "url": url},
         )
         raise CloudflareError(status, _first_error(payload), payload=payload)
      return payload

   def get_zone_by_name(self, zone_name: str) -> Optional[Dict[str, Any]]:
      result = self._request("GET", "/zones", params={"name": zone_name})
      zones: Iterable[Dict[str, Any]] = result.get("result", [])
      return next(iter(zones), None)

   def get_zone_details(self, zone_id: str) -> Dict[str, Any]:
      return self._request("GET", f"/zones/{zone_id}")

   def purge_cache(self, zone_id: str, *, everything: bool = True, files: Optional[list[str]] = None,
               tags: Optional[list[str]] = None, hosts: Optional[list[str]] = None) -> Dict[str, Any]:
      payload: Dict[str, Any]
      if everything:
         payload = {"purge_everything": True}
      else:
         payload = {}
         if files:
            payload["files"] = files
         if tags:
            payload["tags"] = tags
         if hosts:
            payload["hosts"] = hosts
         if not payload:
            raise ValueError("Informe pelo menos um alvo ao nao limpar tudo.")
      return self._request("POST", f"/zones/{zone_id}/purge_cache", body=payload)

   def list_dns_records(self, zone_id: str, *, record_type: Optional[str] = None,
                   name: Optional[str] = None, content: Optional[str] = None, proxied: Optional[bool] = None
                   ) -> Dict[str, Any]:
      params: Dict[str, Any] = {}
      if record_type:
         params["type"] = record_type
      if name:
         params["name"] = name
      if content:
         params["content"] = content
      if proxied is not None:
         params["proxied"] = json.dumps(proxied)
      return self._request("GET", f"/zones/{zone_id}/dns_records", params=params)

   def create_dns_record(self, zone_id: str, *, record_type: str, name: str, content: str,
                    ttl: int = 1, proxied: bool = False, priority: Optional[int] = None,
                    data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
      payload: Dict[str, Any] = {
         "type": record_type,
         "name": name,
         "content": content,
         "ttl": ttl,
         "proxied": proxied,
      }
      if priority is not None:
         payload["priority"] = priority
      if data:
         payload["data"] = data
      return self._request("POST", f"/zones/{zone_id}/dns_records", body=payload)

   def update_dns_record(self, zone_id: str, record_id: str, *, record_type: Optional[str] = None,
                    name: Optional[str] = None, content: Optional[str] = None, ttl: Optional[int] = None,
                    proxied: Optional[bool] = None, priority: Optional[int] = None,
                    data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
      payload: Dict[str, Any] = {}
      if record_type is not None:
         payload["type"] = record_type
      if name is not None:
         payload["name"] = name
      if content is not None:
         payload["content"] = content
      if ttl is not None:
         payload["ttl"] = ttl
      if proxied is not None:
         payload["proxied"] = proxied
      if priority is not None:
         payload["priority"] = priority
      if data is not None:
         payload["data"] = data
      if not payload:
         raise ValueError("Nada para atualizar; informe ao menos um campo.")
      return self._request("PUT", f"/zones/{zone_id}/dns_records/{record_id}", body=payload)

   def delete_dns_record(self, zone_id: str, record_id: str) -> Dict[str, Any]:
      return self._request("DELETE", f"/zones/{zone_id}/dns_records/{record_id}")


def _first_error(payload: Dict[str, Any]) -> str:
   errors = payload.get("errors") or []
   if not errors:
      return payload.get("message", "Erro desconhecido")
   message = errors[0]
   if isinstance(message, dict):
      return message.get("message") or json.dumps(message)
   return str(message)


def _parse_payload(raw: bytes, content_type: str) -> Dict[str, Any]:
   text = raw.decode("utf-8") if raw else ""
   if "application/json" in content_type:
      if not text:
         return {"success": True}
      try:
         return json.loads(text)
      except json.JSONDecodeError:
         return {"success": False, "errors": [{"message": text}]}
   if not text:
      return {"success": True}
   return {"success": False, "errors": [{"message": text}]}


def _configure_logging(verbose: bool) -> None:
   os.makedirs(os.path.dirname(LOG_PATH), exist_ok=True)
   level = logging.DEBUG if verbose else logging.INFO
   logging.basicConfig(
      level=level,
      format="%(asctime)s [%(levelname)s] %(message)s",
      handlers=[
         logging.FileHandler(LOG_PATH, encoding="utf-8"),
         logging.StreamHandler(sys.stderr),
      ],
   )


def _load_from_env() -> CloudflareCredentials:
   return CloudflareCredentials(
      api_token=os.getenv("CLOUDFLARE_API_TOKEN"),
      api_key=os.getenv("CLOUDFLARE_API_KEY"),
      email=os.getenv("CLOUDFLARE_EMAIL"),
      zone_id=os.getenv("CLOUDFLARE_ZONE_ID"),
      account_id=os.getenv("CLOUDFLARE_ACCOUNT_ID"),
   )


def _load_from_args(args: argparse.Namespace) -> CloudflareCredentials:
   return CloudflareCredentials(
      api_token=args.token,
      api_key=args.api_key,
      email=args.email,
      zone_id=args.zone_id,
      account_id=args.account_id,
   )


def _merge_credentials(primary: CloudflareCredentials, secondary: CloudflareCredentials) -> CloudflareCredentials:
   return CloudflareCredentials(
      api_token=primary.api_token or secondary.api_token,
      api_key=primary.api_key or secondary.api_key,
      email=primary.email or secondary.email,
      zone_id=primary.zone_id or secondary.zone_id,
      account_id=primary.account_id or secondary.account_id,
   )


def _load_from_1password(item: str, vault: Optional[str]) -> CloudflareCredentials:
   if shutil.which("op") is None:
      logging.debug("CLI do 1Password nao encontrada; ignorando busca de segredos.")
      return CloudflareCredentials()
   command = ["op", "item", "get", item, "--format", "json"]
   if vault:
      command.extend(["--vault", vault])
   try:
      result = subprocess.run(command, check=True, capture_output=True, text=True)
   except (FileNotFoundError, subprocess.CalledProcessError) as exc:
      logging.warning("Nao foi possivel ler segredos do 1Password", exc_info=exc)
      return CloudflareCredentials()
   data = json.loads(result.stdout)
   fields = _extract_op_fields(data)
   return CloudflareCredentials(
      api_token=fields.get("apiToken"),
      api_key=fields.get("apiKey"),
      email=fields.get("email"),
      zone_id=fields.get("zoneId"),
      account_id=fields.get("accountId"),
   )


def _extract_op_fields(item: Dict[str, Any]) -> Dict[str, str]:
   collected: Dict[str, str] = {}
   for field in item.get("fields", []):
      label = field.get("label") or field.get("id")
      value = field.get("value")
      if label and value:
         collected[label] = value
   for section in item.get("sections", []):
      for field in section.get("fields", []):
         label = field.get("label") or field.get("id")
         value = field.get("value")
         if label and value:
            collected[label] = value
   return collected


def resolve_credentials(args: argparse.Namespace) -> CloudflareCredentials:
   creds = _load_from_args(args)
   env_creds = _load_from_env()
   creds = _merge_credentials(creds, env_creds)
   if args.skip_1password:
      return creds
   op_creds = _load_from_1password(args.op_item or DEFAULT_OP_ITEM, args.op_vault or DEFAULT_OP_VAULT)
   creds = _merge_credentials(creds, op_creds)
   if not creds.api_token and not (creds.api_key and creds.email):
      raise ValueError("Credenciais ausentes; use --token ou --api-key/--email, ou configure no 1Password.")
   return creds


def resolve_zone_id(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> str:
   if args.zone_id:
      return args.zone_id
   if creds.zone_id and not args.zone:
      return creds.zone_id
   if not args.zone:
      raise ValueError("Informe --zone ou --zone-id para identificar a zona Cloudflare.")
   zone = client.get_zone_by_name(args.zone)
   if not zone:
      raise ValueError(f"Zona '{args.zone}' nao encontrada ou inacessivel.")
   return zone["id"]


def cmd_zone(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   if args.zone_id:
      return client.get_zone_details(args.zone_id)
   if args.zone:
      zone = client.get_zone_by_name(args.zone)
      if not zone:
         raise ValueError(f"Zona '{args.zone}' nao encontrada ou inacessivel.")
      return {"result": zone, "success": True}
   if creds.zone_id:
      return client.get_zone_details(creds.zone_id)
   raise ValueError("Use --zone ou --zone-id para inspecionar uma zona.")


def cmd_purge(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   zone_id = resolve_zone_id(client, creds, args)
   return client.purge_cache(
      zone_id,
      everything=args.everything,
      files=args.files,
      tags=args.tags,
      hosts=args.hosts,
   )


def cmd_dns_list(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   zone_id = resolve_zone_id(client, creds, args)
   return client.list_dns_records(
      zone_id,
      record_type=args.type,
      name=args.name,
      content=args.content,
      proxied=args.proxied,
   )


def cmd_dns_create(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   zone_id = resolve_zone_id(client, creds, args)
   data = json.loads(args.data) if args.data else None
   return client.create_dns_record(
      zone_id,
      record_type=args.type,
      name=args.name,
      content=args.content,
      ttl=args.ttl,
      proxied=args.proxied,
      priority=args.priority,
      data=data,
   )


def cmd_dns_update(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   zone_id = resolve_zone_id(client, creds, args)
   data = json.loads(args.data) if args.data else None
   return client.update_dns_record(
      zone_id,
      args.record_id,
      record_type=args.type,
      name=args.name,
      content=args.content,
      ttl=args.ttl,
      proxied=args.proxied,
      priority=args.priority,
      data=data,
   )


def cmd_dns_delete(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
   zone_id = resolve_zone_id(client, creds, args)
   return client.delete_dns_record(zone_id, args.record_id)


def build_parser() -> argparse.ArgumentParser:
   parser = argparse.ArgumentParser(description="Utilitario de automacao para Cloudflare")
   parser.add_argument("--token", help="Token de API da Cloudflare")
   parser.add_argument("--api-key", help="Chave global de API da Cloudflare")
   parser.add_argument("--email", help="E-mail da conta (obrigatorio com --api-key)")
   parser.add_argument("--zone", help="Nome da zona (ex.: otimiza.pro)")
   parser.add_argument("--zone-id", help="Identificador da zona")
   parser.add_argument("--account-id", help="Identificador da conta Cloudflare")
   parser.add_argument("--op-item", default=DEFAULT_OP_ITEM, help="Nome do item no 1Password (padrao: Cloudflare_API)")
   parser.add_argument("--op-vault", default=DEFAULT_OP_VAULT, help="Nome do cofre 1Password (padrao: OtimizaPro)")
   parser.add_argument("--skip-1password", action="store_true", help="Ignora leitura no 1Password")
   parser.add_argument("--verbose", action="store_true", help="Ativa logs de depuracao")

   subparsers = parser.add_subparsers(dest="command", required=True)

   zone_parser = subparsers.add_parser("zone", help="Consultar detalhes da zona")
   zone_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   zone_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   zone_parser.set_defaults(func=cmd_zone)

   purge_parser = subparsers.add_parser("purge", help="Limpar cache da Cloudflare")
   purge_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   purge_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   purge_parser.add_argument("--everything", action="store_true", default=True, help="Limpa todo o cache (padrao)")
   purge_parser.add_argument("--files", nargs="*", help="Lista de arquivos para limpar")
   purge_parser.add_argument("--tags", nargs="*", help="Lista de tags de cache para limpar")
   purge_parser.add_argument("--hosts", nargs="*", help="Lista de hosts para limpar")
   purge_parser.set_defaults(func=cmd_purge)

   dns_parser = subparsers.add_parser("dns", help="Gerenciar registros DNS")
   dns_subparsers = dns_parser.add_subparsers(dest="dns_command", required=True)

   dns_list_parser = dns_subparsers.add_parser("list", help="Listar registros DNS")
   dns_list_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   dns_list_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   dns_list_parser.add_argument("--type", help="Filtrar por tipo de registro (A, CNAME, TXT, ...)")
   dns_list_parser.add_argument("--name", help="Filtrar por nome totalmente qualificado")
   dns_list_parser.add_argument("--content", help="Filtrar por conteudo")
   dns_list_parser.add_argument("--proxied", type=_parse_bool, help="Filtrar por flag proxied (true/false)")
   dns_list_parser.set_defaults(func=cmd_dns_list)

   dns_create_parser = dns_subparsers.add_parser("create", help="Criar registro DNS")
   dns_create_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   dns_create_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   dns_create_parser.add_argument("--type", required=True, help="Tipo do registro")
   dns_create_parser.add_argument("--name", required=True, help="Nome do registro")
   dns_create_parser.add_argument("--content", required=True, help="Conteudo do registro")
   dns_create_parser.add_argument("--ttl", type=int, default=1, help="TTL do registro (1 = auto)")
   dns_create_parser.add_argument("--proxied", action="store_true", help="Define registro como proxied")
   dns_create_parser.add_argument("--priority", type=int, help="Prioridade (MX/SRV)")
   dns_create_parser.add_argument("--data", help="Dados adicionais em JSON")
   dns_create_parser.set_defaults(func=cmd_dns_create)

   dns_update_parser = dns_subparsers.add_parser("update", help="Atualizar registro DNS")
   dns_update_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   dns_update_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   dns_update_parser.add_argument("record_id", help="Identificador do registro DNS")
   dns_update_parser.add_argument("--type", help="Novo tipo")
   dns_update_parser.add_argument("--name", help="Novo nome")
   dns_update_parser.add_argument("--content", help="Novo conteudo")
   dns_update_parser.add_argument("--ttl", type=int, help="Novo TTL")
   dns_update_parser.add_argument("--proxied", type=_parse_bool, help="Define flag proxied (true/false)")
   dns_update_parser.add_argument("--priority", type=int, help="Prioridade")
   dns_update_parser.add_argument("--data", help="Dados adicionais em JSON")
   dns_update_parser.set_defaults(func=cmd_dns_update)

   dns_delete_parser = dns_subparsers.add_parser("delete", help="Excluir registro DNS")
   dns_delete_parser.add_argument("--zone", help="Nome da zona (sobrepoe --zone global)")
   dns_delete_parser.add_argument("--zone-id", help="ID da zona (sobrepoe --zone-id global)")
   dns_delete_parser.add_argument("record_id", help="Identificador do registro DNS a excluir")
   dns_delete_parser.set_defaults(func=cmd_dns_delete)

   return parser


def _parse_bool(value: str) -> bool:
   truthy = {"1", "true", "t", "yes", "y"}
   falsy = {"0", "false", "f", "no", "n"}
   value_normalized = value.strip().lower()
   if value_normalized in truthy:
      return True
   if value_normalized in falsy:
      return False
   raise argparse.ArgumentTypeError("Valor booleano esperado (true/false).")


def main(argv: Optional[Sequence[str]] = None) -> int:
   parser = build_parser()
   args = parser.parse_args(argv)
   _configure_logging(args.verbose)
   try:
      creds = resolve_credentials(args)
      client = CloudflareClient(creds)
      result = args.func(client, creds, args)
      print(json.dumps(result, indent=2))
      return 0
   except ValueError as exc:
      logging.error(str(exc))
      print(str(exc), file=sys.stderr)
      return 1
   except CloudflareError as exc:
      logging.error("Erro retornado pela Cloudflare", exc_info=exc)
      payload = exc.payload or {"errors": [{"message": str(exc)}]}
      print(json.dumps(payload, indent=2), file=sys.stderr)
      return 2
   except (ConnectionError, subprocess.SubprocessError) as exc:
      logging.error("Falha de execucao", exc_info=exc)
      print(f"Falha de execucao: {exc}", file=sys.stderr)
      return 3


if __name__ == "__main__":
   sys.exit(main())

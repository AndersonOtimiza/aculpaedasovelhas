"""Cloudflare automation helper.

This module provides a small CLI that wraps the Cloudflare REST API and mirrors
common operations described in the original guide: zone lookup, cache purge, and
DNS record management. Credentials are resolved from 1Password (CLI) when
available, with fallbacks to explicit CLI arguments and environment variables.

Usage examples:

    python cloudflare.py zone --zone otimiza.pro
    python cloudflare.py purge --zone otimiza.pro
    python cloudflare.py dns list --zone otimiza.pro
    python cloudflare.py dns create --zone otimiza.pro --type A --name portal \
        --content 200.160.1.1 --proxied

Set the following environment variables or leave them blank to rely on the
1Password item named ``Cloudflare_API`` in the ``OtimizaPro`` vault:

    CLOUDFLARE_API_TOKEN
    CLOUDFLARE_API_KEY
    CLOUDFLARE_EMAIL
    CLOUDFLARE_ZONE_ID

The script prefers API tokens and falls back to global API key if required.
"""
from __future__ import annotations

import argparse
import json
import logging
import os
import shutil
import subprocess
import sys
from dataclasses import dataclass
from typing import Any, Dict, Iterable, Optional

import requests

API_BASE_URL = "https://api.cloudflare.com/client/v4"
DEFAULT_OP_ITEM = "Cloudflare_API"
DEFAULT_OP_VAULT = "OtimizaPro"
LOG_PATH = os.path.join("logs", "cloudflare.log")


class CloudflareError(RuntimeError):
    """Raised when the Cloudflare API returns an error response."""

    def __init__(self, status_code: int, message: str, *, payload: Optional[Dict[str, Any]] = None) -> None:
        super().__init__(f"Cloudflare API error {status_code}: {message}")
        self.status_code = status_code
        self.payload = payload or {}


@dataclass
class CloudflareCredentials:
    """Represents authentication material and scoped configuration."""

    api_token: Optional[str] = None
    api_key: Optional[str] = None
    email: Optional[str] = None
    zone_id: Optional[str] = None
    account_id: Optional[str] = None

    def headers(self) -> Dict[str, str]:
        if self.api_token:
            return {
                "Authorization": f"Bearer {self.api_token}",
                "Content-Type": "application/json",
            }
        if self.api_key and self.email:
            return {
                "X-Auth-Key": self.api_key,
                "X-Auth-Email": self.email,
                "Content-Type": "application/json",
            }
        raise ValueError("Missing authentication; provide an API token or API key/email pair.")


class CloudflareClient:
    """Thin wrapper around the Cloudflare REST API."""

    def __init__(self, credentials: CloudflareCredentials, *, timeout: int = 30) -> None:
        self._credentials = credentials
        self._timeout = timeout

    def _request(self, method: str, path: str, **kwargs: Any) -> Dict[str, Any]:
        url = f"{API_BASE_URL}{path}"
        headers = kwargs.pop("headers", {})
        headers.update(self._credentials.headers())
        response = requests.request(method, url, headers=headers, timeout=self._timeout, **kwargs)
        content_type = response.headers.get("Content-Type", "")
        payload: Dict[str, Any]
        if "application/json" in content_type:
            payload = response.json()
        else:
            payload = {"success": response.ok, "errors": [{"message": response.text}]}
        if not response.ok or not payload.get("success", True):
            logging.error("Cloudflare request failed", extra={"status": response.status_code, "payload": payload})
            raise CloudflareError(response.status_code, _first_error(payload), payload=payload)
        return payload

    def get_zone_by_name(self, zone_name: str) -> Optional[Dict[str, Any]]:
        result = self._request("GET", "/zones", params={"name": zone_name})
        zones: Iterable[Dict[str, Any]] = result.get("result", [])
        return next(iter(zones), None)

    def get_zone_details(self, zone_id: str) -> Dict[str, Any]:
        return self._request("GET", f"/zones/{zone_id}")

    def purge_cache(self, zone_id: str, *, everything: bool = True, files: Optional[list[str]] = None,
                    tags: Optional[list[str]] = None, hosts: Optional[list[str]] = None) -> Dict[str, Any]:
        payload: Dict[str, Any]
        if everything:
            payload = {"purge_everything": True}
        else:
            payload = {}
            if files:
                payload["files"] = files
            if tags:
                payload["tags"] = tags
            if hosts:
                payload["hosts"] = hosts
            if not payload:
                raise ValueError("Specify at least one target when not purging everything.")
        return self._request("POST", f"/zones/{zone_id}/purge_cache", json=payload)

    def list_dns_records(self, zone_id: str, *, record_type: Optional[str] = None,
                         name: Optional[str] = None, content: Optional[str] = None, proxied: Optional[bool] = None
                         ) -> Dict[str, Any]:
        params: Dict[str, Any] = {}
        if record_type:
            params["type"] = record_type
        if name:
            params["name"] = name
        if content:
            params["content"] = content
        if proxied is not None:
            params["proxied"] = json.dumps(proxied)
        return self._request("GET", f"/zones/{zone_id}/dns_records", params=params)

    def create_dns_record(self, zone_id: str, *, record_type: str, name: str, content: str,
                          ttl: int = 1, proxied: bool = False, priority: Optional[int] = None,
                          data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        payload: Dict[str, Any] = {
            "type": record_type,
            "name": name,
            "content": content,
            "ttl": ttl,
            "proxied": proxied,
        }
        if priority is not None:
            payload["priority"] = priority
        if data:
            payload["data"] = data
        return self._request("POST", f"/zones/{zone_id}/dns_records", json=payload)

    def update_dns_record(self, zone_id: str, record_id: str, *, record_type: Optional[str] = None,
                          name: Optional[str] = None, content: Optional[str] = None, ttl: Optional[int] = None,
                          proxied: Optional[bool] = None, priority: Optional[int] = None,
                          data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        payload: Dict[str, Any] = {}
        if record_type is not None:
            payload["type"] = record_type
        if name is not None:
            payload["name"] = name
        if content is not None:
            payload["content"] = content
        if ttl is not None:
            payload["ttl"] = ttl
        if proxied is not None:
            payload["proxied"] = proxied
        if priority is not None:
            payload["priority"] = priority
        if data is not None:
            payload["data"] = data
        if not payload:
            raise ValueError("Nothing to update; provide at least one field.")
        return self._request("PUT", f"/zones/{zone_id}/dns_records/{record_id}", json=payload)

    def delete_dns_record(self, zone_id: str, record_id: str) -> Dict[str, Any]:
        return self._request("DELETE", f"/zones/{zone_id}/dns_records/{record_id}")


def _first_error(payload: Dict[str, Any]) -> str:
    errors = payload.get("errors") or []
    if not errors:
        return payload.get("message", "Unknown error")
    message = errors[0]
    if isinstance(message, dict):
        return message.get("message") or json.dumps(message)
    return str(message)


def _configure_logging(verbose: bool) -> None:
    os.makedirs(os.path.dirname(LOG_PATH), exist_ok=True)
    level = logging.DEBUG if verbose else logging.INFO
    logging.basicConfig(
        level=level,
        format="%(asctime)s [%(levelname)s] %(message)s",
        handlers=[
            logging.FileHandler(LOG_PATH, encoding="utf-8"),
            logging.StreamHandler(sys.stderr),
        ],
    )


def _load_from_env() -> CloudflareCredentials:
    return CloudflareCredentials(
        api_token=os.getenv("CLOUDFLARE_API_TOKEN"),
        api_key=os.getenv("CLOUDFLARE_API_KEY"),
        email=os.getenv("CLOUDFLARE_EMAIL"),
        zone_id=os.getenv("CLOUDFLARE_ZONE_ID"),
        account_id=os.getenv("CLOUDFLARE_ACCOUNT_ID"),
    )


def _load_from_args(args: argparse.Namespace) -> CloudflareCredentials:
    return CloudflareCredentials(
        api_token=args.token,
        api_key=args.api_key,
        email=args.email,
        zone_id=args.zone_id,
        account_id=args.account_id,
    )


def _merge_credentials(primary: CloudflareCredentials, secondary: CloudflareCredentials) -> CloudflareCredentials:
    return CloudflareCredentials(
        api_token=primary.api_token or secondary.api_token,
        api_key=primary.api_key or secondary.api_key,
        email=primary.email or secondary.email,
        zone_id=primary.zone_id or secondary.zone_id,
        account_id=primary.account_id or secondary.account_id,
    )


def _load_from_1password(item: str, vault: Optional[str]) -> CloudflareCredentials:
    if shutil.which("op") is None:
        logging.debug("1Password CLI not found; skipping secret lookup.")
        return CloudflareCredentials()
    command = ["op", "item", "get", item, "--format", "json"]
    if vault:
        command.extend(["--vault", vault])
    try:
        result = subprocess.run(command, check=True, capture_output=True, text=True)
    except (FileNotFoundError, subprocess.CalledProcessError) as exc:
        logging.warning("Unable to read secrets from 1Password", exc_info=exc)
        return CloudflareCredentials()
    data = json.loads(result.stdout)
    fields = _extract_op_fields(data)
    return CloudflareCredentials(
        api_token=fields.get("apiToken"),
        api_key=fields.get("apiKey"),
        email=fields.get("email"),
        zone_id=fields.get("zoneId"),
        account_id=fields.get("accountId"),
    )


def _extract_op_fields(item: Dict[str, Any]) -> Dict[str, str]:
    collected: Dict[str, str] = {}
    for field in item.get("fields", []):
        label = field.get("label") or field.get("id")
        value = field.get("value")
        if label and value:
            collected[label] = value
    for section in item.get("sections", []):
        for field in section.get("fields", []):
            label = field.get("label") or field.get("id")
            value = field.get("value")
            if label and value:
                collected[label] = value
    return collected


def resolve_credentials(args: argparse.Namespace) -> CloudflareCredentials:
    creds = _load_from_args(args)
    env_creds = _load_from_env()
    creds = _merge_credentials(creds, env_creds)
    if args.skip_1password:
        return creds
    op_creds = _load_from_1password(args.op_item or DEFAULT_OP_ITEM, args.op_vault or DEFAULT_OP_VAULT)
    creds = _merge_credentials(creds, op_creds)
    if not creds.api_token and not (creds.api_key and creds.email):
        raise ValueError("Authentication details missing; use --token or --api-key/--email, or configure 1Password.")
    return creds


def resolve_zone_id(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> str:
    if args.zone_id:
        return args.zone_id
    if creds.zone_id and not args.zone:
        return creds.zone_id
    if not args.zone:
        raise ValueError("Specify --zone or --zone-id to identify the Cloudflare zone.")
    zone = client.get_zone_by_name(args.zone)
    if not zone:
        raise ValueError(f"Zone '{args.zone}' not found or not accessible.")
    return zone["id"]


def cmd_zone(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    if args.zone_id:
        return client.get_zone_details(args.zone_id)
    if args.zone:
        zone = client.get_zone_by_name(args.zone)
        if not zone:
            raise ValueError(f"Zone '{args.zone}' not found or not accessible.")
        return {"result": zone, "success": True}
    if creds.zone_id:
        return client.get_zone_details(creds.zone_id)
    raise ValueError("Provide --zone or --zone-id to inspect a zone.")


def cmd_purge(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    zone_id = resolve_zone_id(client, creds, args)
    return client.purge_cache(
        zone_id,
        everything=args.everything,
        files=args.files,
        tags=args.tags,
        hosts=args.hosts,
    )


def cmd_dns_list(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    zone_id = resolve_zone_id(client, creds, args)
    return client.list_dns_records(
        zone_id,
        record_type=args.type,
        name=args.name,
        content=args.content,
        proxied=args.proxied,
    )


def cmd_dns_create(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    zone_id = resolve_zone_id(client, creds, args)
    data = json.loads(args.data) if args.data else None
    return client.create_dns_record(
        zone_id,
        record_type=args.type,
        name=args.name,
        content=args.content,
        ttl=args.ttl,
        proxied=args.proxied,
        priority=args.priority,
        data=data,
    )


def cmd_dns_update(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    zone_id = resolve_zone_id(client, creds, args)
    data = json.loads(args.data) if args.data else None
    return client.update_dns_record(
        zone_id,
        args.record_id,
        record_type=args.type,
        name=args.name,
        content=args.content,
        ttl=args.ttl,
        proxied=args.proxied,
        priority=args.priority,
        data=data,
    )


def cmd_dns_delete(client: CloudflareClient, creds: CloudflareCredentials, args: argparse.Namespace) -> Dict[str, Any]:
    zone_id = resolve_zone_id(client, creds, args)
    return client.delete_dns_record(zone_id, args.record_id)


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="Cloudflare API automation helper")
    parser.add_argument("--token", help="Cloudflare API token")
    parser.add_argument("--api-key", help="Cloudflare global API key")
    parser.add_argument("--email", help="Cloudflare account email (required with --api-key)")
    parser.add_argument("--zone", help="Zone name (eg. otimiza.pro)")
    parser.add_argument("--zone-id", help="Cloudflare zone identifier")
    parser.add_argument("--account-id", help="Cloudflare account identifier")
    parser.add_argument("--op-item", default=DEFAULT_OP_ITEM, help="1Password item name (default: Cloudflare_API)")
    parser.add_argument("--op-vault", default=DEFAULT_OP_VAULT, help="1Password vault name (default: OtimizaPro)")
    parser.add_argument("--skip-1password", action="store_true", help="Disable 1Password lookup")
    parser.add_argument("--verbose", action="store_true", help="Enable debug logging")

    subparsers = parser.add_subparsers(dest="command", required=True)

    zone_parser = subparsers.add_parser("zone", help="Inspect zone details")
    zone_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    zone_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    zone_parser.set_defaults(func=cmd_zone)

    purge_parser = subparsers.add_parser("purge", help="Purge Cloudflare cache")
    purge_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    purge_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    purge_parser.add_argument("--everything", action="store_true", default=True, help="Purge entire cache (default)")
    purge_parser.add_argument("--files", nargs="*", help="List of files to purge")
    purge_parser.add_argument("--tags", nargs="*", help="List of cache tags to purge")
    purge_parser.add_argument("--hosts", nargs="*", help="List of hosts to purge")
    purge_parser.set_defaults(func=cmd_purge)

    dns_parser = subparsers.add_parser("dns", help="Manage DNS records")
    dns_subparsers = dns_parser.add_subparsers(dest="dns_command", required=True)

    dns_list_parser = dns_subparsers.add_parser("list", help="List DNS records")
    dns_list_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    dns_list_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    dns_list_parser.add_argument("--type", help="Record type filter (A, CNAME, TXT, ...)")
    dns_list_parser.add_argument("--name", help="Fully qualified record name filter")
    dns_list_parser.add_argument("--content", help="Content filter")
    dns_list_parser.add_argument("--proxied", type=_parse_bool, help="Filter by proxied flag (true/false)")
    dns_list_parser.set_defaults(func=cmd_dns_list)

    dns_create_parser = dns_subparsers.add_parser("create", help="Create DNS record")
    dns_create_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    dns_create_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    dns_create_parser.add_argument("--type", required=True, help="Record type")
    dns_create_parser.add_argument("--name", required=True, help="Record name")
    dns_create_parser.add_argument("--content", required=True, help="Record content")
    dns_create_parser.add_argument("--ttl", type=int, default=1, help="Record TTL (default: 1 = auto)")
    dns_create_parser.add_argument("--proxied", action="store_true", help="Set record as proxied")
    dns_create_parser.add_argument("--priority", type=int, help="Record priority (MX/SRV)")
    dns_create_parser.add_argument("--data", help="Additional record data as JSON string")
    dns_create_parser.set_defaults(func=cmd_dns_create)

    dns_update_parser = dns_subparsers.add_parser("update", help="Update DNS record")
    dns_update_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    dns_update_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    dns_update_parser.add_argument("record_id", help="Identifier of the DNS record to update")
    dns_update_parser.add_argument("--type", help="New record type")
    dns_update_parser.add_argument("--name", help="New record name")
    dns_update_parser.add_argument("--content", help="New record content")
    dns_update_parser.add_argument("--ttl", type=int, help="New TTL value")
    dns_update_parser.add_argument("--proxied", type=_parse_bool, help="Set proxied flag (true/false)")
    dns_update_parser.add_argument("--priority", type=int, help="Record priority")
    dns_update_parser.add_argument("--data", help="Additional record data as JSON string")
    dns_update_parser.set_defaults(func=cmd_dns_update)

    dns_delete_parser = dns_subparsers.add_parser("delete", help="Delete DNS record")
    dns_delete_parser.add_argument("--zone", help="Zone name (fallback to global --zone)")
    dns_delete_parser.add_argument("--zone-id", help="Zone id (fallback to global --zone-id)")
    dns_delete_parser.add_argument("record_id", help="Identifier of the DNS record to delete")
    dns_delete_parser.set_defaults(func=cmd_dns_delete)

    return parser


def _parse_bool(value: str) -> bool:
    truthy = {"1", "true", "t", "yes", "y"}
    falsy = {"0", "false", "f", "no", "n"}
    value_normalized = value.strip().lower()
    if value_normalized in truthy:
        return True
    if value_normalized in falsy:
        return False
    raise argparse.ArgumentTypeError("Boolean value expected (true/false).")


def main(argv: Optional[Iterable[str]] = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    _configure_logging(args.verbose)
    try:
        creds = resolve_credentials(args)
        client = CloudflareClient(creds)
        result = args.func(client, creds, args)
        print(json.dumps(result, indent=2))
        return 0
    except ValueError as exc:
        logging.error(str(exc))
        print(str(exc), file=sys.stderr)
        return 1
    except CloudflareError as exc:
        logging.error("Cloudflare error", exc_info=exc)
        payload = exc.payload or {"errors": [{"message": str(exc)}]}
        print(json.dumps(payload, indent=2), file=sys.stderr)
        return 2
    except (requests.RequestException, subprocess.SubprocessError) as exc:
        logging.error("Runtime failure", exc_info=exc)
        print(f"Runtime failure: {exc}", file=sys.stderr)
        return 3


if __name__ == "__main__":
    sys.exit(main())

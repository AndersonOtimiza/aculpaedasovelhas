#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Matriz Belem, AnC — revelação de sinais e hashes intertextuais bíblicos
Projeto em bloco único (1 arquivo) para rodar local.

Recursos:
- SQLite + FTS5 (busca literal em texto normalizado)
- Normalização (hebraico/aramaico/grego)
- Hash_EXATO (SHA-256), SHINGLE_k, MINHASH (caseiro), SIMHASH (caseiro)
- Query por alvo texto/número (texto: FTS; número: literal "7"/"sete"/"seven")
- Saída em "matriz" Markdown no terminal

Uso rápido:
  python mtc_app.py init --db bible.db
  python mtc_app.py ingest --db bible.db --corpus he --tsv data/hebrew.tsv --bookcol 0 --chapcol 1 --versecol 2 --textcol 3
  python mtc_app.py ingest --db bible.db --corpus gr --tsv data/greek.tsv  --bookcol 0 --chapcol 1 --versecol 2 --textcol 3
  python mtc_app.py index  --db bible.db --k 3 --perms 64
  python mtc_app.py query  --db bible.db --alvo "sete cabeças" --k 3 --top 80

TSV esperado (por padrão):
  col0=book, col1=chapter, col2=verse, col3=text_raw
Delimitador padrão: TAB. Ajuste com --sep se necessário.

Observação:
- Sem base de tokens/lemmas/Strong’s, o modo "âncoras por lema" não roda. Aqui é literal+hashes+similaridade.
"""

from __future__ import annotations

import argparse
import hashlib
import sqlite3
import unicodedata
from dataclasses import dataclass
from typing import Iterable, List, Tuple, Optional, Dict, Any

try:
    import regex as re  # melhor unicode
except Exception:  # fallback
    import re  # type: ignore


# -------------------------
# Normalização Unicode
# -------------------------

_HEBREW_MARKS_RE = re.compile(r"[\p{Mn}]") if hasattr(re, "compile") else None  # Mn = combining marks
_GREEK_MARKS_RE = _HEBREW_MARKS_RE

def _nfkd(s: str) -> str:
    return unicodedata.normalize("NFKD", s)

def normalize_hebrew_aramaic(s: str, strip_marks: bool = True) -> str:
    s = _nfkd(s)
    if strip_marks and _HEBREW_MARKS_RE is not None:
        s = _HEBREW_MARKS_RE.sub("", s)
    s = re.sub(r"\s+", " ", s).strip()
    return s

def normalize_greek(s: str, strip_marks: bool = True) -> str:
    s = _nfkd(s)
    if strip_marks and _GREEK_MARKS_RE is not None:
        s = _GREEK_MARKS_RE.sub("", s)
    s = s.replace("ς", "σ")
    s = re.sub(r"\s+", " ", s).strip()
    return s

def normalize_generic(s: str) -> str:
    s = _nfkd(s)
    s = re.sub(r"\s+", " ", s).strip()
    return s

def normalize_by_corpus(corpus: str, text_raw: str, strip_marks: bool = True) -> str:
    c = corpus.lower().strip()
    if c in ("he", "heb", "hebrew", "arc", "aramaic", "aram"):
        return normalize_hebrew_aramaic(text_raw, strip_marks=strip_marks)
    if c in ("gr", "greek", "gk"):
        return normalize_greek(text_raw, strip_marks=strip_marks)
    return normalize_generic(text_raw)


# -------------------------
# Tokenização / shingles
# -------------------------

_WORD_RE = re.compile(r"\p{L}+\p{M}*|\d+|[^\s]") if hasattr(re, "compile") else re.compile(r"\w+|\d+|[^\s]")

def tokenize(s: str) -> List[str]:
    return [m.group(0) for m in _WORD_RE.finditer(s)]

def shingles(tokens: List[str], k: int) -> List[Tuple[str, ...]]:
    if k <= 0:
        return []
    if len(tokens) < k:
        return []
    return [tuple(tokens[i:i+k]) for i in range(0, len(tokens) - k + 1)]

def shingle_strings(tokens: List[str], k: int) -> List[str]:
    return [" ".join(t) for t in shingles(tokens, k)]


# -------------------------
# Hashes
# -------------------------

def sha256_hex(s: str) -> str:
    return hashlib.sha256(s.encode("utf-8")).hexdigest()

def _hash64(s: str, seed: int) -> int:
    # 64-bit hash determinístico via blake2b
    h = hashlib.blake2b(digest_size=8, person=seed.to_bytes(4, "little", signed=False))
    h.update(s.encode("utf-8"))
    return int.from_bytes(h.digest(), "little", signed=False)

def minhash_signature(items: Iterable[str], perms: int = 64) -> List[int]:
    # MinHash caseiro: para cada perm (seed), pega o mínimo hash64
    # itens vazios => assinatura de zeros
    items = list(items)
    sig: List[int] = []
    for seed in range(perms):
        m = None
        for it in items:
            v = _hash64(it, seed)
            m = v if m is None else (v if v < m else m)
        sig.append(m if m is not None else 0)
    return sig

def minhash_similarity(sig_a: List[int], sig_b: List[int]) -> float:
    if not sig_a or not sig_b or len(sig_a) != len(sig_b):
        return 0.0
    eq = 0
    for a, b in zip(sig_a, sig_b):
        if a == b:
            eq += 1
    return eq / len(sig_a)

def simhash64(items: Iterable[str]) -> int:
    # SimHash 64-bit por itens (strings). Cada item contribui com hash64 seed fixo=999.
    # Sem pesos explícitos.
    v = [0] * 64
    for it in items:
        h = _hash64(it, 999)
        for i in range(64):
            bit = (h >> i) & 1
            v[i] += 1 if bit else -1
    out = 0
    for i in range(64):
        if v[i] > 0:
            out |= (1 << i)
    return out

def hamming64(a: int, b: int) -> int:
    return (a ^ b).bit_count()

def simhash_similarity(a: int, b: int) -> float:
    # 1 - normalized Hamming distance
    return 1.0 - (hamming64(a, b) / 64.0)


# -------------------------
# DB
# -------------------------

SCHEMA_SQL = """
PRAGMA journal_mode=WAL;
PRAGMA synchronous=NORMAL;

CREATE TABLE IF NOT EXISTS verses (
  corpus    TEXT NOT NULL,
  book      TEXT NOT NULL,
  chapter   INTEGER NOT NULL,
  verse     INTEGER NOT NULL,
  text_raw  TEXT NOT NULL,
  text_norm TEXT NOT NULL,
  PRIMARY KEY (corpus, book, chapter, verse)
);

-- FTS5: texto normalizado
CREATE VIRTUAL TABLE IF NOT EXISTS verses_fts USING fts5(
  corpus UNINDEXED,
  book   UNINDEXED,
  chapter UNINDEXED,
  verse  UNINDEXED,
  text_norm,
  content='verses',
  content_rowid='rowid'
);

-- hashes por verso
CREATE TABLE IF NOT EXISTS verse_hashes (
  corpus    TEXT NOT NULL,
  book      TEXT NOT NULL,
  chapter   INTEGER NOT NULL,
  verse     INTEGER NOT NULL,
  sha256_norm TEXT NOT NULL,
  simhash64   INTEGER NOT NULL,
  minhash_sig BLOB NOT NULL,  -- sequência de uint64 (little-endian)
  k          INTEGER NOT NULL,
  perms      INTEGER NOT NULL,
  PRIMARY KEY (corpus, book, chapter, verse)
);

CREATE INDEX IF NOT EXISTS idx_verses_book ON verses(corpus, book, chapter, verse);
CREATE INDEX IF NOT EXISTS idx_hash_sha ON verse_hashes(corpus, sha256_norm);
"""

def connect(db_path: str) -> sqlite3.Connection:
    con = sqlite3.connect(db_path)
    con.row_factory = sqlite3.Row
    return con

def init_db(db_path: str) -> None:
    con = connect(db_path)
    try:
        con.executescript(SCHEMA_SQL)
        con.commit()
    finally:
        con.close()

def rebuild_fts(con: sqlite3.Connection) -> None:
    # Rebuild do content table -> fts
    con.execute("INSERT INTO verses_fts(verses_fts) VALUES('rebuild')")
    con.commit()

def upsert_verse(con: sqlite3.Connection, corpus: str, book: str, chapter: int, verse: int, raw: str, norm: str) -> None:
    con.execute(
        """
        INSERT INTO verses(corpus, book, chapter, verse, text_raw, text_norm)
        VALUES(?,?,?,?,?,?)
        ON CONFLICT(corpus, book, chapter, verse) DO UPDATE SET
          text_raw=excluded.text_raw,
          text_norm=excluded.text_norm
        """,
        (corpus, book, chapter, verse, raw, norm),
    )

def upsert_fts_from_verses(con: sqlite3.Connection) -> None:
    # mantém fts sincronizado: apaga e recarrega (simplicidade)
    con.execute("DELETE FROM verses_fts")
    con.execute(
        """
        INSERT INTO verses_fts(rowid, corpus, book, chapter, verse, text_norm)
        SELECT rowid, corpus, book, chapter, verse, text_norm
        FROM verses
        """
    )
    con.commit()

def encode_sig_u64(sig: List[int]) -> bytes:
    return b"".join(int(x).to_bytes(8, "little", signed=False) for x in sig)

def decode_sig_u64(blob: bytes, perms: int) -> List[int]:
    out: List[int] = []
    for i in range(perms):
        start = i * 8
        out.append(int.from_bytes(blob[start:start+8], "little", signed=False))
    return out


# -------------------------
# Matriz (linhas)
# -------------------------

@dataclass
class MatrixRow:
    id: int
    a_ref: str
    b_ref: str
    categoria: str
    sinal: str
    det: int
    evidencia_minima: str
    ancoras: str
    hash_tipo: str
    hash_valor: str
    score_total: int

def ref(corpus: str, book: str, chapter: int, verse: int) -> str:
    return f"{corpus}:{book} {chapter}:{verse}"

def hash_mtc(a_ref: str, b_ref: str, anchors: List[str], se: str, k: int, limiar: float) -> str:
    anchors_s = ",".join(anchors[:5])
    return f"MTC:{a_ref}->{b_ref}|A:{{{anchors_s}}}|S:{se}|k:{k}|t:{limiar:.2f}"


# -------------------------
# Scoring (simples e auditável)
# -------------------------

def score_pair(
    shared_phrase: bool,
    shared_shingles: int,
    minhash_sim: float,
    simhash_sim: float,
    has_number: bool,
    has_connective: bool,
    has_structure_flag: bool,
) -> int:
    score = 0
    # Lexical
    if shared_phrase:
        score += 20
    if shared_shingles >= 2:
        score += 10
    elif shared_shingles == 1:
        score += 5

    # Similaridade
    if minhash_sim >= 0.82:
        score += 20
    elif minhash_sim >= 0.70:
        score += 10

    if simhash_sim >= 0.82:
        score += 10
    elif simhash_sim >= 0.70:
        score += 5

    # Numérico / Conectores / Estrutura (flags)
    if has_number:
        score += 10
    if has_connective:
        score += 5
    if has_structure_flag:
        score += 10

    return min(score, 100)

def classify(score: int) -> str:
    if score >= 60:
        return "forte"
    if score >= 30:
        return "provavel"
    return "fraco"


# -------------------------
# Ingest / Index / Query
# -------------------------

def ingest_tsv(
    db: str,
    corpus: str,
    tsv_path: str,
    bookcol: int,
    chapcol: int,
    versecol: int,
    textcol: int,
    sep: str = "\t",
    strip_marks: bool = True,
    limit: Optional[int] = None,
) -> None:
    con = connect(db)
    try:
        with open(tsv_path, "r", encoding="utf-8") as f:
            n = 0
            for line in f:
                if limit is not None and n >= limit:
                    break
                line = line.rstrip("\n")
                if not line:
                    continue
                parts = line.split(sep)
                if max(bookcol, chapcol, versecol, textcol) >= len(parts):
                    continue
                book = parts[bookcol].strip()
                chapter = int(parts[chapcol].strip())
                verse = int(parts[versecol].strip())
                raw = parts[textcol]
                norm = normalize_by_corpus(corpus, raw, strip_marks=strip_marks)
                upsert_verse(con, corpus, book, chapter, verse, raw, norm)
                n += 1
            con.commit()
        upsert_fts_from_verses(con)
        rebuild_fts(con)
    finally:
        con.close()

def index_hashes(db: str, k: int = 3, perms: int = 64, batch: int = 2000) -> None:
    con = connect(db)
    try:
        cur = con.execute("SELECT corpus, book, chapter, verse, text_norm FROM verses")
        rows = cur.fetchall()

        n = 0
        for r in rows:
            corpus = r["corpus"]
            book = r["book"]
            chapter = int(r["chapter"])
            verse = int(r["verse"])
            text_norm = r["text_norm"]

            toks = tokenize(text_norm)
            sh = shingle_strings(toks, k=k)

            sha = sha256_hex(text_norm)
            mh = minhash_signature(sh, perms=perms)
            sh64 = simhash64(sh)

            con.execute(
                """
                INSERT INTO verse_hashes(corpus, book, chapter, verse, sha256_norm, simhash64, minhash_sig, k, perms)
                VALUES(?,?,?,?,?,?,?,?,?)
                ON CONFLICT(corpus, book, chapter, verse) DO UPDATE SET
                  sha256_norm=excluded.sha256_norm,
                  simhash64=excluded.simhash64,
                  minhash_sig=excluded.minhash_sig,
                  k=excluded.k,
                  perms=excluded.perms
                """,
                (corpus, book, chapter, verse, sha, int(sh64), encode_sig_u64(mh), k, perms),
            )
            n += 1
            if n % batch == 0:
                con.commit()
        con.commit()
    finally:
        con.close()

def expand_numeric_targets(s: str) -> List[str]:
    out = [s]
    s_low = s.strip().lower()
    if s_low in ("7", "sete", "seven"):
        out.extend(["7", "sete", "seven"])
    if s_low in ("666",):
        out.extend(["666"])
    return sorted(set(out))

def query_candidates_fts(con: sqlite3.Connection, alvo: str, top: int = 50) -> List[sqlite3.Row]:
    terms = expand_numeric_targets(alvo)
    qparts = []
    for t in terms:
        t = t.replace('"', '""')
        qparts.append(f'"{t}"')
    q = " OR ".join(qparts) if qparts else f'"{alvo}"'
    cur = con.execute(
        """
        SELECT corpus, book, chapter, verse, text_raw, text_norm, bm25(verses_fts) AS rank
        FROM verses_fts
        WHERE verses_fts MATCH ?
        ORDER BY rank
        LIMIT ?
        """,
        (q, top),
    )
    return cur.fetchall()

def load_hash(con: sqlite3.Connection, corpus: str, book: str, chapter: int, verse: int) -> Optional[sqlite3.Row]:
    cur = con.execute(
        """
        SELECT sha256_norm, simhash64, minhash_sig, k, perms
        FROM verse_hashes
        WHERE corpus=? AND book=? AND chapter=? AND verse=?
        """,
        (corpus, book, chapter, verse),
    )
    return cur.fetchone()

def compute_shared_shingles(a_norm: str, b_norm: str, k: int) -> int:
    a = set(shingle_strings(tokenize(a_norm), k))
    b = set(shingle_strings(tokenize(b_norm), k))
    if not a or not b:
        return 0
    return len(a & b)

def has_shared_phrase(a_norm: str, b_norm: str) -> bool:
    a = shingle_strings(tokenize(a_norm), 3)
    b = set(shingle_strings(tokenize(b_norm), 3))
    return any(x in b for x in a)

def detect_connective(alvo: str, a_norm: str, b_norm: str) -> bool:
    term = alvo.strip()
    if not term:
        return False
    return (term in a_norm) and (term in b_norm)

def detect_number(alvo: str, a_norm: str, b_norm: str) -> bool:
    terms = expand_numeric_targets(alvo)
    for t in terms:
        if t.isdigit() and (t in a_norm) and (t in b_norm):
            return True
    return False

def query_matrix(
    db: str,
    alvo: str,
    k: int = 3,
    top: int = 80,
    limiar_minhash: float = 0.82,
    limiar_simhash: float = 0.82,
    max_pairs: int = 200,
) -> List[MatrixRow]:
    con = connect(db)
    try:
        cands = query_candidates_fts(con, alvo=alvo, top=top)
        if len(cands) < 2:
            return []

        enriched: List[Dict[str, Any]] = []
        for r in cands:
            h = load_hash(con, r["corpus"], r["book"], int(r["chapter"]), int(r["verse"]))
            if not h:
                continue
            enriched.append({"row": r, "hash": h})

        out: List[MatrixRow] = []
        pair_id = 1
        n = len(enriched)
        cap = 0

        for i in range(n):
            for j in range(i + 1, n):
                if cap >= max_pairs:
                    break

                A = enriched[i]["row"]
                B = enriched[j]["row"]
                hA = enriched[i]["hash"]
                hB = enriched[j]["hash"]

                permsA = int(hA["perms"])
                permsB = int(hB["perms"])
                if permsA != permsB:
                    continue

                sigA = decode_sig_u64(hA["minhash_sig"], permsA)
                sigB = decode_sig_u64(hB["minhash_sig"], permsB)
                mh_sim = minhash_similarity(sigA, sigB)

                shA = int(hA["simhash64"])
                shB = int(hB["simhash64"])
                sh_sim = simhash_similarity(shA, shB)

                shared_k = compute_shared_shingles(A["text_norm"], B["text_norm"], k=k)
                phrase = has_shared_phrase(A["text_norm"], B["text_norm"])

                has_num = detect_number(alvo, A["text_norm"], B["text_norm"])
                has_conn = detect_connective(alvo, A["text_norm"], B["text_norm"])
                has_struct = False  # reservado (inclusio/quiasmo) — plugar depois

                score = score_pair(
                    shared_phrase=phrase,
                    shared_shingles=shared_k,
                    minhash_sim=mh_sim,
                    simhash_sim=sh_sim,
                    has_number=has_num,
                    has_connective=has_conn,
                    has_structure_flag=has_struct,
                )

                if score < 30 and mh_sim < 0.70 and shared_k == 0 and not phrase:
                    continue

                det = 2 if (mh_sim >= limiar_minhash or sh_sim >= limiar_simhash or phrase or shared_k >= 2) else 1

                a_ref = ref(A["corpus"], A["book"], int(A["chapter"]), int(A["verse"]))
                b_ref = ref(B["corpus"], B["book"], int(B["chapter"]), int(B["verse"]))

                if phrase or shared_k >= 2:
                    categoria = "II"
                    sinal = "Shared-phrase / Shared shingles"
                elif mh_sim >= 0.70 or sh_sim >= 0.70:
                    categoria = "EIC"
                    sinal = "Echo by similarity hashes"
                else:
                    categoria = "MT"
                    sinal = "Lexical proximity"

                evidencia = f"shared_k={shared_k}; minhash={mh_sim:.2f}; simhash={sh_sim:.2f}; score={score} ({classify(score)})"
                anchors = ""
                mtc = hash_mtc(a_ref, b_ref, anchors=[], se="NA", k=k, limiar=limiar_minhash)

                out.append(MatrixRow(
                    id=pair_id,
                    a_ref=a_ref,
                    b_ref=b_ref,
                    categoria=categoria,
                    sinal=sinal,
                    det=det,
                    evidencia_minima=evidencia,
                    ancoras=anchors,
                    hash_tipo="HASH_MTC",
                    hash_valor=mtc,
                    score_total=score,
                ))
                pair_id += 1
                cap += 1

            if cap >= max_pairs:
                break

        out.sort(key=lambda r: r.score_total, reverse=True)
        return out
    finally:
        con.close()

def print_matrix_md(rows: List[MatrixRow], alvo: str, show: int = 50) -> None:
    print(f"# Matriz Belem, AnC — alvo: {alvo}")
    print("")
    if not rows:
        print("Sem achados (ou falta rodar index).")
        return
    print("| ID | A_ref | B_ref | Cat | Sinal | 0/1/2 | Evidência mínima | Hash_tipo | Hash_valor | Score |")
    print("|---:|---|---|---|---|---:|---|---|---|---:|")
    for r in rows[:show]:
        ev = r.evidencia_minima.replace("|", "\\|")
        hv = r.hash_valor.replace("|", "\\|")
        print(f"| {r.id} | {r.a_ref} | {r.b_ref} | {r.categoria} | {r.sinal} | {r.det} | {ev} | {r.hash_tipo} | {hv} | {r.score_total} |")
    print("")


# -------------------------
# CLI
# -------------------------

def main(argv: Optional[List[str]] = None) -> int:
    p = argparse.ArgumentParser(prog="mtc_app.py", add_help=True)
    sub = p.add_subparsers(dest="cmd", required=True)

    p_init = sub.add_parser("init", help="Cria o banco e o schema")
    p_init.add_argument("--db", required=True)

    p_ing = sub.add_parser("ingest", help="Importa TSV para verses + FTS")
    p_ing.add_argument("--db", required=True)
    p_ing.add_argument("--corpus", required=True, choices=["he", "arc", "gr"])
    p_ing.add_argument("--tsv", required=True)
    p_ing.add_argument("--sep", default="\\t", help="Delimitador. Use \\t para TAB.")
    p_ing.add_argument("--bookcol", type=int, default=0)
    p_ing.add_argument("--chapcol", type=int, default=1)
    p_ing.add_argument("--versecol", type=int, default=2)
    p_ing.add_argument("--textcol", type=int, default=3)
    p_ing.add_argument("--keep-marks", action="store_true", help="Não remover marcas/acentos (he/gr)")
    p_ing.add_argument("--limit", type=int, default=None)

    p_idx = sub.add_parser("index", help="Gera hashes (sha/minhash/simhash) para todos os versos")
    p_idx.add_argument("--db", required=True)
    p_idx.add_argument("--k", type=int, default=3)
    p_idx.add_argument("--perms", type=int, default=64)

    p_q = sub.add_parser("query", help="Consulta por alvo e gera matriz (pares entre candidatos)")
    p_q.add_argument("--db", required=True)
    p_q.add_argument("--alvo", required=True)
    p_q.add_argument("--k", type=int, default=3)
    p_q.add_argument("--top", type=int, default=80)
    p_q.add_argument("--max-pairs", type=int, default=200)
    p_q.add_argument("--limiar-minhash", type=float, default=0.82)
    p_q.add_argument("--limiar-simhash", type=float, default=0.82)
    p_q.add_argument("--show", type=int, default=50)

    args = p.parse_args(argv)

    if args.cmd == "init":
        init_db(args.db)
        print("OK: db inicializado")
        return 0

    if args.cmd == "ingest":
        sep = args.sep.encode("utf-8").decode("unicode_escape")
        ingest_tsv(
            db=args.db,
            corpus=args.corpus,
            tsv_path=args.tsv,
            bookcol=args.bookcol,
            chapcol=args.chapcol,
            versecol=args.versecol,
            textcol=args.textcol,
            sep=sep,
            strip_marks=not args.keep_marks,
            limit=args.limit,
        )
        print("OK: ingest completo + FTS rebuild")
        return 0

    if args.cmd == "index":
        index_hashes(db=args.db, k=args.k, perms=args.perms)
        print("OK: index de hashes completo")
        return 0

    if args.cmd == "query":
        rows = query_matrix(
            db=args.db,
            alvo=args.alvo,
            k=args.k,
            top=args.top,
            limiar_minhash=args.limiar_minhash,
            limiar_simhash=args.limiar_simhash,
            max_pairs=args.max_pairs,
        )
        print_matrix_md(rows, alvo=args.alvo, show=args.show)
        return 0

    return 2

if __name__ == "__main__":
    raise SystemExit(main())

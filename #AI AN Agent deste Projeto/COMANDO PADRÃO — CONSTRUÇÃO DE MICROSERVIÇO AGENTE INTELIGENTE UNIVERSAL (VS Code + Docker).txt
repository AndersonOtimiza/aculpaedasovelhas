# 9) app/main.py — API do agente
@"
import os
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from dotenv import load_dotenv

from app.core.intent import IntentRouter
from app.core.registry import Registry
from app.core.policy import Policy

load_dotenv()

app = FastAPI(title="Agente Universal", version="1.0.0")

router  = IntentRouter()
policy  = Policy()
registry= Registry(policy=policy)

class CommandIn(BaseModel):
    text: str
    target: str | None = None
    adapter: str | None = None
    args: dict | None = None
    dry_run: bool | None = None

@app.get("/health")
def health():
    return { "status": "ok", "adapters": list(registry.adapters.keys()) }

@app.post("/command")
def command(ci: CommandIn):
    try:
        plan = router.plan(ci.text, ci.target, ci.adapter)
        # Merge args fornecidos
        if ci.args:
            plan["args"] = { **plan.get("args", {}), **ci.args }
        # Dry run prioritário
        dry = ci.dry_run if ci.dry_run is not None else (os.getenv("DRY_RUN","false").lower() == "true")

        adapter_name = plan.get("adapter")
        action       = plan.get("action")
        args         = plan.get("args", {})
        target       = plan.get("target")

        adapter = registry.get(adapter_name)
        if not adapter:
            return {"ok": False, "message": f"Adapter não encontrado: {adapter_name}", "plan": plan}

        # Enrich args com dados do target (base_url, auth, paths)
        target_conf = registry.get_target(target)
        if target_conf:
            args = registry.enrich_with_target(args, target_conf)

        # Checagem de políticas
        policy.validate(adapter_name, action, args, bypass=os.getenv("ALLOWLIST_BYPASS","false").lower()=="true")

        if dry:
            return {"ok": True, "dry_run": True, "plan": plan, "args": args}

        result = adapter.execute(action, **args)
        return {"ok": True, "plan": plan, "result": result}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
"@ | Set-Content -Encoding UTF8 -Path "$Root\agent\app\main.py"

# 10) app/core/intent.py — roteador de intenção (LLM + heurística)

import os, json, re
from typing import Dict, Optional

def _llm_available() -> bool:
    prov = os.getenv(""LLM_PROVIDER"", """").lower()
    if prov == ""ollama"": return bool(os.getenv(""OLLAMA_BASE_URL""))
    if prov == ""openai"": return bool(os.getenv(""OPENAI_API_KEY""))
    if prov == ""azure_openai"": return bool(os.getenv(""AZURE_OPENAI_API_KEY"")) and bool(os.getenv(""AZURE_OPENAI_ENDPOINT""))
    return False

def _call_llm(prompt: str) -> str:
    provider = os.getenv(""LLM_PROVIDER"", ""ollama"").lower()
    model    = os.getenv(""MODEL"", ""llama3.1:instruct"")
    try:
        from openai import OpenAI
        if provider == ""ollama"":
            client = OpenAI(base_url=os.getenv(""OLLAMA_BASE_URL""), api_key=""ollama"")
        elif provider == ""azure_openai"":
            from openai import AzureOpenAI
            client = AzureOpenAI(
                api_key=os.getenv(""AZURE_OPENAI_API_KEY""),
                api_version=""2024-08-01-preview"",
                azure_endpoint=os.getenv(""AZURE_OPENAI_ENDPOINT""),
            )
        else:
            client = OpenAI(api_key=os.getenv(""OPENAI_API_KEY""))

        resp = client.chat.completions.create(
            model=model,
            messages=[
                {""role"":""system"",""content"":""Você é um planejador. Responda apenas JSON com: {adapter, action, target?, args{}}. Adapters: http_rest, shell_cli, browser_playwright. Seja determinístico."" },
                {""role"":""user"",""content"":prompt}
            ],
            temperature=0.1
        )
        return resp.choices[0].message.content or """"
    except Exception:
        return """"

class IntentRouter:
    def plan(self, text: str, target: Optional[str]=None, adapter: Optional[str]=None) -> Dict:
        # Se usuário já especificou adapter/target, respeitar e usar ação inferida
        if _llm_available():
            out = _call_llm(f""Texto: {text}\nTarget preferido: {target}\nAdapter preferido: {adapter}"")
            try:
                j = json.loads(out.strip(""` \n""))
                if isinstance(j, dict) and ""adapter"" in j and ""action"" in j:
                    if target: j.setdefault(""target"", target)
                    if adapter: j[""adapter""] = adapter
                    return j
            except Exception:
                pass

        # Heurística offline
        t = text.lower()
        chosen_adapter = adapter
        chosen_action  = ""execute""
        args = {}

        # Palavras-chave -> adapter
        if not chosen_adapter:
            if any(k in t for k in [""curl"", ""api"", ""http"", ""get "", ""post "", ""put "", ""delete "", ""graphql""]):
                chosen_adapter = ""http_rest""
            elif any(k in t for k in [""navegar"",""abrir"",""clicar"",""preencher"",""login"",""screenshot"",""ui"",""interface""]):
                chosen_adapter = ""browser_playwright""
            elif any(k in t for k in [""executar"",""rodar"",""cli"",""compilar"",""build"",""deploy"",""docker"",""npm"",""git"",""kubectl""]):
                chosen_adapter = ""shell_cli""
            else:
                chosen_adapter = ""browser_playwright""

        # Ações comuns
        if chosen_adapter == ""http_rest"":
            # extrai método
            method = ""get""
            for m in [""get"",""post"",""put"",""delete""]:
                if re.search(rf""\\b{m}\\b"", t): method = m; break
            url = """"  # pode vir do target
            murl = re.search(r""https?://\\S+"", text)
            if murl: url = murl.group(0)
            args = { ""method"": method.upper(), ""url"": url, ""headers"": {}, ""body"": {} }
            chosen_action = ""request""

        elif chosen_adapter == ""browser_playwright"":
            # ação em passos (navegar/clicar/digitar)
            args = {
                ""steps"": [
                    { ""op"": ""navigate"", ""url"": """" },
                    # o detalhamento pode ser enriquecido por LLM em execuções futuras
                ],
                ""headless"": True
            }
            chosen_action = ""run_steps""

        elif chosen_adapter == ""shell_cli"":
            # comando simples inferido do texto
            cmd = text
            args = { ""cmd"": cmd, ""cwd"": ""/workspace"" }
            chosen_action = ""run""

        return { ""adapter"": chosen_adapter, ""action"": chosen_action, ""target"": target or ""default"", ""args"": args }
"@ | Set-Content -Encoding UTF8 -Path "$Root\agent\app\core\intent.py"

# 11) app/core/policy.py — aplicação de políticas/allowlist

import os, json, re

class Policy:
    def __init__(self, path: str = ""/app/policies/allowlist.json""):
        self.path = path
        try:
            with open(self.path, ""r"", encoding=""utf-8"") as f:
                self.rules = json.load(f)
        except Exception:
            self.rules = {}

    def validate(self, adapter: str, action: str, args: dict, bypass: bool=False):
        if bypass: 
            return True
        r = self.rules.get(adapter, {})
        if adapter == ""shell_cli"":
            cmd = (args or {}).get(""cmd"", """").strip()
            denies = r.get(""deny"", [])
            for patt in denies:
                if re.search(patt, cmd): 
                    raise ValueError(f""Comando bloqueado por deny: {patt}"")
            allows = r.get(""allowedCommands"", [])
            if allows:
                allowed = any(re.search(p, cmd) for p in allows)
                if not allowed:
                    raise ValueError(""Comando shell não permitido pelo allowlist."")
        elif adapter == ""http_rest"":
            url = (args or {}).get(""url"", """")
            domains = r.get(""allowedDomains"", [])
            if domains and url:
                allowed = any(re.search(p, url) for p in domains)
                if not allowed:
                    raise ValueError(""Domínio HTTP não permitido pelo allowlist."")
        elif adapter == ""browser_playwright"":
            steps = (args or {}).get(""steps"", [])
            max_steps = int(r.get(""maxSteps"", 25))
            if len(steps) > max_steps:
                raise ValueError(f""Excesso de passos no fluxo de UI: {len(steps)} > {max_steps}"")
        return True
"@ | Set-Content -Encoding UTF8 -Path "$Root\agent\app\core\policy.py"

import json, os
from typing import Dict, Any

from app.adapters.http_rest.adapter import HTTPAdapter
from app.adapters.browser_playwright.adapter import BrowserAdapter
from app.adapters.shell_cli.adapter import ShellAdapter

class Registry:
    def __init__(self, policy):
        self.policy = policy
        self.adapters = {
            ""http_rest"": HTTPAdapter(),
            ""browser_playwright"": BrowserAdapter(),
            ""shell_cli"": ShellAdapter()
        }
        self.targets = self._load_targets()

    def _load_targets(self) -> Dict[str, Any]:
        path = ""/app/targets/targets.json""
        try:
            with open(path, ""r"", encoding=""utf-8"") as f:
                return json.load(f)
        except Exception:
            return { ""default"": { ""adapter"": ""browser_playwright"", ""base_url"": ""https://example.com"" } }

    def get(self, name: str):
        return self.adapters.get(name)

    def get_target(self, name: str | None):
        if not name: return None
        return self.targets.get(name)

    def enrich_with_target(self, args: dict, target_conf: dict) -> dict:
        out = dict(args or {})
        # Disponibiliza base_url e auth
        if ""base_url"" in target_conf:
            out.setdefault(""base_url"", target_conf[""base_url""])
            # Preenche URL vazia se existir path relativo
            if out.get(""url"", """") == """" and target_conf.get(""base_url""):
                out[""url""] = target_conf[""base_url""]
        auth = target_conf.get(""auth"", {})
        if auth.get(""type"") == ""basic_env"":
            from_base = (os.getenv(auth.get(""user_env"", """")), os.getenv(auth.get(""pass_env"", """")))
            out[""basic_auth""] = { ""user"": from_base[0] or """", ""pass"": from_base[1] or """" }
        elif auth.get(""type"") == ""bearer_env"":
            token = os.getenv(auth.get(""token_env"", """")) or """"
            out[""bearer_token""] = token
        # Base path para shell
        if target_conf.get(""adapter"") == ""shell_cli"" and target_conf.get(""base_path""):
            out.setdefault(""cwd"", target_conf[""base_path""])
        return out

import requests, json
from typing import Dict, Any

class HTTPAdapter:
    def actions(self):
        return [""request""]

    def execute(self, action: str, **kwargs):
        if action != ""request"": 
            return {""ok"": False, ""message"": f""Ação não suportada: {action}""}
        method = (kwargs.get(""method"",""GET"") or ""GET"").upper()
        url    = kwargs.get(""url"") or kwargs.get(""base_url"")
        headers= kwargs.get(""headers"") or {}
        body   = kwargs.get(""body"") or {}
        # auth
        if ""basic_auth"" in kwargs and kwargs[""basic_auth""].get(""user""):
            u = kwargs[""basic_auth""][""user""]; p = kwargs[""basic_auth""][""pass""]
            auth = (u,p)
        else:
            auth = None
        if ""bearer_token"" in kwargs and kwargs[""bearer_token""]:
            headers = { **headers, ""Authorization"": f""Bearer {kwargs['bearer_token']}"" }

        resp = requests.request(method, url, headers=headers, json=body if method != ""GET"" else None, timeout=90, auth=auth)
        try:
            data = resp.json()
        except Exception:
            data = { ""text"": resp.text }
        return { ""status"": resp.status_code, ""data"": data, ""headers"": dict(resp.headers) }

import os
from typing import List, Dict, Any
from playwright.sync_api import sync_playwright

class BrowserAdapter:
    def actions(self):
        return [""run_steps""]

    def execute(self, action: str, **kwargs):
        if action != ""run_steps"":
            return {""ok"": False, ""message"": f""Ação não suportada: {action}""}
        steps: List[Dict[str, Any]] = kwargs.get(""steps"", [])
        headless = bool(kwargs.get(""headless"", True))
        base_url = kwargs.get(""base_url"", """")
        results = []

        with sync_playwright() as p:
            browser = p.chromium.launch(headless=headless)
            context = browser.new_context()
            page = context.new_page()

            for s in steps:
                op = s.get(""op"")
                if op == ""navigate"":
                    url = s.get(""url"") or base_url
                    page.goto(url, wait_until=""domcontentloaded"")
                    results.append({""op"":op,""status"":""ok"",""url"":page.url})
                elif op == ""click"":
                    sel = s.get(""selector"")
                    page.click(sel)
                    results.append({""op"":op,""selector"":sel,""status"":""ok""})
                elif op == ""type"":
                    sel = s.get(""selector""); val = s.get(""value"", """")
                    page.fill(sel, val)
                    results.append({""op"":op,""selector"":sel,""status"":""ok""})
                elif op == ""press"":
                    sel = s.get(""selector""); key = s.get(""key"",""Enter"")
                    page.press(sel, key)
                    results.append({""op"":op,""selector"":sel,""key"":key,""status"":""ok""})
                elif op == ""wait_for"":
                    sel = s.get(""selector""); timeout = int(s.get(""timeout"", 30000))
                    page.wait_for_selector(sel, timeout=timeout)
                    results.append({""op"":op,""selector"":sel,""status"":""ok""})
                elif op == ""screenshot"":
                    path = s.get(""path"",""/workspace/screenshot.png"")
                    page.screenshot(path=path, full_page=True)
                    results.append({""op"":op,""path"":path,""status"":""ok""})
                else:
                    results.append({""op"":op,""status"":""ignored"",""reason"":""op desconhecida""})

            context.close(); browser.close()
        return { ""steps"": steps, ""results"": results }
"@ | Set-Content -Encoding UTF8 -Path "$Root\agent\app\adapters\browser_playwright\adapter.py"

import subprocess, shlex, os

class ShellAdapter:
    def actions(self):
        return [""run""]

    def execute(self, action: str, **kwargs):
        if action != ""run"": 
            return {""ok"": False, ""message"": f""Ação não suportada: {action}""}
        cmd = kwargs.get(""cmd"")
        cwd = kwargs.get(""cwd"", ""/workspace"")
        env = os.environ.copy()
        extra_env = kwargs.get(""env"", {})
        env.update({ k:str(v) for k,v in extra_env.items() })

        p = subprocess.run(cmd if isinstance(cmd, list) else shlex.split(str(cmd)), cwd=cwd, capture_output=True, text=True, env=env, shell=False)
        return { ""returncode"": p.returncode, ""stdout"": p.stdout, ""stderr"": p.stderr }
